export default {
  async fetch(req, env) {
    const url = new URL(req.url);

    // --- CORS preflight ---
    if (req.method === 'OPTIONS') {
      return new Response(null, { headers: corsHeaders() });
    }

    // ===== ROUTES =====
    
    // Luồng ClickBank (Giữ nguyên cho cũ)
    if (url.pathname === "/thank-you") {
      return thankyouPage(url, env, "CLICKBANK");
    }

    // Luồng Digistore24 (Mới - Dùng cho trang /thank-you-ds/)
    if (url.pathname === "/thank-you-ds") {
      return thankyouPage(url, env, "DIGISTORE24");
    }

    if (url.pathname === "/issue" && req.method === "POST") {
      return issueToken(req, env);
    }

    if (url.pathname === "/download") {
      return downloadFile(url, env);
    }

    return new Response("Not found", { status: 404 });
  }
};

/** ============ CONFIG & HELPERS ============ **/
const TTL_MINUTES = 60 * 24; 

function randomId() {
  return crypto.randomUUID().replace(/-/g, "");
}

function corsHeaders() {
  return {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type',
    'Access-Control-Max-Age': '86400',
  };
}

async function getTitle(env, sku) {
  if (env.PRODUCT_TITLES) {
    const t = await env.PRODUCT_TITLES.get(sku);
    if (t) return t;
  }
  return sku.replace(/_/g, " ");
}

async function getFilename(env, sku) {
  if (env.EBOOK_PRODUCTS) {
    return await env.EBOOK_PRODUCTS.get(sku);
  }
  return null;
}

/** ============ TOKEN & DOWNLOAD LOGIC (DÙNG CHUNG) ============ **/
async function issueToken(req, env) {
  const { orderId, email, sku } = await req.json();
  const filename = await getFilename(env, sku);
  
  if (!filename) return Response.json({ error: "Product file not found" }, { status: 400 });

  const token = randomId();
  const exp = Date.now() + TTL_MINUTES * 60 * 1000;
  
  await env.TOKENS.put("t:"+token, JSON.stringify({ orderId, email, sku, filename, exp }), { expirationTtl: TTL_MINUTES*60 });

  const url = new URL("/download", req.url);
  url.searchParams.set("t", token);
  return Response.json({ url: url.toString() });
}

async function downloadFile(url, env) {
  const t = url.searchParams.get("t");
  const raw = await env.TOKENS.get("t:"+t);
  if (!raw) return new Response("Link expired", { status: 410 });

  const meta = JSON.parse(raw);
  const obj = await env.EBOOKS_BUCKET.get(meta.filename);
  if (!obj) return new Response("File not found", { status: 404 });

  const headers = new Headers();
  headers.set("Content-Type", "application/zip");
  headers.set("Content-Disposition", `attachment; filename="${meta.filename.split('/').pop()}"`);
  return new Response(obj.body, { headers });
}
