<!-- 1. BẮT MÃ AFFILIATE VÀ LƯU VÀO BỘ NHỚ (VẠN NĂNG) -->
<script data-no-optimize="1" data-no-defer="1">
(function() {
    const params = new URLSearchParams(window.location.search);
    const affID = params.get('aff');

    if (params.has('jv')) {
        localStorage.setItem('aff_source', 'jvzoo');
    } else if (params.has('wh')) {
        localStorage.setItem('aff_source', 'whop');
        if (affID) localStorage.setItem('whop_id', affID);
    } else if (params.has('cc')) {
        localStorage.setItem('aff_source', 'copecart');
        if (affID) localStorage.setItem('copecart_id', affID);
    } else if (params.has('h') || params.has('hotmart')) {
        localStorage.setItem('aff_source', 'hotmart');
        if (affID) localStorage.setItem('hotmart_id', affID);
    } else if (affID) {
        // Mặc định nếu chỉ có aff= thì là Digistore24
        localStorage.setItem('aff_source', 'ds24');
        localStorage.setItem('ds24_aff', affID);
    }
})();
</script>

<!-- 3. LOGIC ĐỔI NÚT MUA HÀNG (CHÍNH XÁC QUY ƯỚC TỪNG SÀN) -->
<script data-no-optimize="1" data-no-defer="1">
(function() {
    const CONFIG = {
        ds24_badge_url: "https://www.digistore24.com/trusted-badge/43187/WFJ04GSnvqXthEg/salespage",
        jv_base: "https://www.jvzoo.com/b/0/431859/1",
        ds24_base: "https://www.checkout-ds24.com/product/658756",
        hotmart_base: "https://pay.hotmart.com/M103669726F",
        whop_base: "https://whop.com/checkout/plan_ZqLqAyfpmwM2N",
        cc_base: "https://copecart.com/us/products/29850120/checkout" // THAY LINK COPECART CỦA ANH VÀO ĐÂY
    };

    function updateBuyButtons() {
        const path = window.location.pathname;
        const source = localStorage.getItem('aff_source');
        const buttons = document.querySelectorAll('.dynamic-buy-button a, a.dynamic-buy-button');

        buttons.forEach(function(btn) {
            let finalLink = "";
            
            if (source === 'whop') {
                const id = localStorage.getItem('whop_id');
                finalLink = CONFIG.whop_base + (id ? "?a=" + id : ""); // Quy ước của Whop là ?a=
            } 
            else if (source === 'copecart') {
                const id = localStorage.getItem('copecart_id');
                finalLink = CONFIG.cc_base + (id ? "?affiliate_id=" + id : ""); // Quy ước của CopeCart
            }
            else if (source === 'jvzoo') {
                finalLink = CONFIG.jv_base;
            }
            else if (source === 'hotmart') {
                // Hotmart thường dùng cookies, nhưng có thể thêm param nếu dùng Hotpay
                finalLink = CONFIG.hotmart_base + "?checkoutMode=10";
            }
            else {
                const id = localStorage.getItem('ds24_aff');
                finalLink = CONFIG.ds24_base + (id ? "?aff=" + id : ""); // Quy ước của DS24
            }
            
            btn.href = finalLink;
        });

        // Hiện Disclaimer cho JVZoo
        const jvDisclaimer = document.querySelector('.jvzoo-disclaimer');
        if (jvDisclaimer) { jvDisclaimer.style.display = (source === 'jvzoo' ? 'block' : 'none'); }
    }

    updateBuyButtons();
    window.addEventListener('load', updateBuyButtons);
    setInterval(updateBuyButtons, 1500);
})();
</script>
