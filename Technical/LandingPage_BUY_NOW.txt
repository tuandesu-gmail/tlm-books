<!-- 1. BẮT MÃ AFFILIATE VÀ LƯU VÀO BỘ NHỚ -->
<script data-no-optimize="1" data-no-defer="1">
(function() {
    const params = new URLSearchParams(window.location.search);
    // Lưu nguồn vào bộ nhớ
    if (params.has('aff')) {
        localStorage.setItem('aff_source', 'ds24');
        localStorage.setItem('ds24_aff', params.get('aff'));
    } else if (params.has('jv')) {
        localStorage.setItem('aff_source', 'jvzoo');
    } else if (params.has('h') || params.has('hotmart')) {
        localStorage.setItem('aff_source', 'hotmart');
    } else if (params.has('wh') || params.has('whop')) {
        localStorage.setItem('aff_source', 'whop');
    } else if (params.has('cc') || params.has('copecart')) { // THÊM LOGIC COPECART
        localStorage.setItem('aff_source', 'copecart');
    }
})();
</script>

<!-- 3. LOGIC ĐỔI NÚT MUA HÀNG DỰA TRÊN BỘ NHỚ -->
<script data-no-optimize="1" data-no-defer="1">
(function() {
    const CONFIG = {
        ds24_badge_url: "https://www.digistore24.com/trusted-badge/43187/WFJ04GSnvqXthEg/salespage",
        jv_base: "https://www.jvzoo.com/b/0/431859/1",
        ds24_base: "https://www.checkout-ds24.com/product/658756",
        hotmart_base: "https://pay.hotmart.com/M103669726F",
        whop_base: "https://whop.com/checkout/plan_ZqLqAyfpmwM2N",
        copecart_base: "https://copecart.com/us/products/29850120/checkout" // LINK CHECKOUT COPECART
    };

    function updateBuyButtons() {
        const path = window.location.pathname;
        
        // Xác định ngôn ngữ
        let currentLang = 'en';
        const langMap = {'/zh-hant/':'zh', '/pt-br/':'pt', '/fr/':'fr', '/de/':'de', '/es/':'es', '/ko/':'ko', '/ja/':'ja', '/vi/':'vi'};
        for (let key in langMap) {
            if (path.includes(key)) { currentLang = langMap[key]; break; }
        }

        const source = localStorage.getItem('aff_source');
        const savedAff = localStorage.getItem('ds24_aff');

        const buttons = document.querySelectorAll('.dynamic-buy-button a, a.dynamic-buy-button');
        
        buttons.forEach(function(btn) {
            if (source === 'whop') {
                btn.href = CONFIG.whop_base;
            } else if (source === 'copecart') {
                // Chuyển đến CopeCart (Hỗ trợ param lang)
                btn.href = CONFIG.copecart_base + "?lang=" + currentLang;
            } else if (source === 'jvzoo') {
                btn.href = CONFIG.jv_base;
            } else if (source === 'hotmart') {
                btn.href = CONFIG.hotmart_base + "?checkoutMode=10&lang=" + currentLang;
            } else {
                // Mặc định là Digistore24
                let dsLink = CONFIG.ds24_base + "?lang=" + currentLang;
                if (savedAff) dsLink += "&aff=" + savedAff;
                btn.href = dsLink;
            }
        });

        // Quản lý JVZoo Disclaimer
        const jvDisclaimer = document.querySelector('.jvzoo-disclaimer');
        if (jvDisclaimer) { 
            jvDisclaimer.style.setProperty("display", (source === 'jvzoo' ? '' : 'none'), "important");
        }

        // Quản lý Badge DS24
        const isLandingPage = path.includes('/books-collection/');
        const shouldShowBadge = (isLandingPage && (source === 'ds24' || !source));
        
        if (shouldShowBadge) {
            if (!document.getElementById('ds24_badge_script')) {
                const badgeScript = document.createElement('script');
                badgeScript.id = 'ds24_badge_script';
                badgeScript.src = CONFIG.ds24_badge_url;
                document.body.appendChild(badgeScript);
            }
        } else {
            const existingBadge = document.getElementById('ds24_trusted_badge');
            if (existingBadge) existingBadge.style.display = 'none';
        }
    }

    updateBuyButtons();
    window.addEventListener('load', updateBuyButtons);
    let c = 0; const i = setInterval(function() { updateBuyButtons(); if (++c > 5) clearInterval(i); }, 1000);
})();
</script>
