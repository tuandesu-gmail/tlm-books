<!-- 1. BẮT MÃ AFFILIATE (Chạy ngay lập tức) -->
<script data-no-optimize="1" data-no-defer="1">
(function() {
    const params = new URLSearchParams(window.location.search);
    if (params.has('aff')) {
        localStorage.setItem('aff_source', 'ds24');
        localStorage.setItem('ds24_aff', params.get('aff'));
    } else if (params.has('hop')) {
        localStorage.setItem('aff_source', 'cb');
        localStorage.setItem('cb_hop', params.get('hop'));
    }
})();
</script>

<!-- 3. LOGIC CẬP NHẬT NÚT BẤM VÀ NGÔN NGỮ (Chạy toàn trang) -->
<script data-no-optimize="1" data-no-defer="1">
(function() {
    const CONFIG = {
        ds24_badge_url: "https://www.digistore24.com/trusted-badge/43187/WFJ04GSnvqXthEg/salespage",
        cb_base: "https://livesmedia.pay.clickbank.net/?cbitems=1",
        ds24_base: "https://www.checkout-ds24.com/product/658756"
    };

    function updateAllContent() {
        const path = window.location.pathname;
        
        // KIỂM TRA XEM CÓ PHẢI LÀ TRANG LANDING PAGE KHÔNG
        const isLandingPage = path.includes('/books-collection/');
        
        let dsLang = 'en';
        if (path.includes('/zh-hant/')) dsLang = 'zh';
        else if (path.includes('/pt-br/')) dsLang = 'pt';
        else if (path.includes('/fr/')) dsLang = 'fr';
        else if (path.includes('/de/')) dsLang = 'de';
        else if (path.includes('/es/')) dsLang = 'es';
        else if (path.includes('/ko/')) dsLang = 'ko';
        else if (path.includes('/ja/')) dsLang = 'ja';
        else if (path.includes('/vi/')) dsLang = 'vi';

        const source = localStorage.getItem('aff_source');
        const savedAff = localStorage.getItem('ds24_aff');
        const savedHop = localStorage.getItem('cb_hop');
        const isCb = (source === 'cb');
        
        // Cập nhật nút bấm (Chỉ chạy nếu tìm thấy nút trên trang)
        const buttons = document.querySelectorAll('.dynamic-buy-button a, a.dynamic-buy-button');
        buttons.forEach(function(btn) {
            if (isCb) {
                let cbLink = CONFIG.cb_base;
                if (savedHop) cbLink += "&vcb=1&affiliate=" + savedHop;
                btn.href = cbLink;
            } else {
                let dsLink = CONFIG.ds24_base + "?lang=" + dsLang;
                if (savedAff) dsLink += "&aff=" + savedAff;
                btn.href = dsLink;
            }
        });

        // Ẩn hiện ClickBank Disclaimer
        const cbDisclaimer = document.querySelector('.clickbank-disclaimer');
        if (cbDisclaimer) { 
            cbDisclaimer.style.setProperty("display", (isCb ? 'block' : 'none'), "important"); 
        }

        // 5. CHỈ HIỆN BADGE NẾU LÀ TRANG LANDING PAGE VÀ LÀ TRAFFIC DS24/ORGANIC
        const isDs24Traffic = (source === 'ds24' || !isCb);
        
        if (isLandingPage && isDs24Traffic) {
            if (!document.getElementById('ds24_badge_script')) {
                const badgeScript = document.createElement('script');
                badgeScript.id = 'ds24_badge_script';
                badgeScript.src = CONFIG.ds24_badge_url;
                document.body.appendChild(badgeScript);
            }
        } else {
            // Nếu không phải landing page, đảm bảo badge không hiển thị (nếu lỡ nạp)
            const existingBadge = document.getElementById('ds24_trusted_badge');
            if (existingBadge) existingBadge.style.display = 'none';
        }
    }

    updateAllContent();
    window.addEventListener('load', updateAllContent);
    let c = 0; const i = setInterval(function() { updateAllContent(); if (++c > 5) clearInterval(i); }, 1000);
})();
</script>

        line-height: 1.5 !important;
    }
